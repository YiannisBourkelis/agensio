per frame copied and per stream acknowledged, a flag per produce and two in enqueue.

- base: f8594a01, bench/tmp/ab/f8594a01/build/agensio; new: the three commits (working tree),

The script:

```bash
#!/usr/bin/env bash
# The h3 /style.css c=64 m=10 row by hand, base and new alternated, with perf stat on the server:
# CPU us per request, instructions and L1 instruction-cache misses per request. Run in
# agensio-devbox-perf as root (--cap-add SYS_ADMIN --cap-add PERFMON).
# usage: h3-row-perf.sh BASE NEW [ROUNDS] [PATH] [CONNS] [STREAMS]
set -euo pipefail
BASE=$1 NEW=$2 ROUNDS=${3:-4} URLPATH=${4:-/style.css} CONNS=${5:-64} STREAMS=${6:-10}
H2LOAD=/opt/nghttp2/bin/h2load
CONF=bench/tmp/ab-agensio-h3.toml
HZ=$(getconf CLK_TCK)
export LC_NUMERIC=C
ulimit -n 65536
one() {  # binary -> "us instr l1i"
  "$1" -c "$CONF" >/dev/null 2>&1 & local pid=$!
  for _ in $(seq 1 50); do curl -sk --http3-only -o /dev/null "https://127.0.0.1:8443$URLPATH" && break; sleep 0.1; done
  $H2LOAD --alpn-list=h3 -D 1 -c "$CONNS" -m "$STREAMS" -t 4 "https://127.0.0.1:8443$URLPATH" >/dev/null 2>&1  # warm
  local c0; c0=$(awk '{print $14+$15}' /proc/$pid/stat)
  perf stat -x, -e instructions,L1-icache-load-misses -p $pid -o /tmp/perf.$$ -- \
    $H2LOAD --alpn-list=h3 -D 5 -c "$CONNS" -m "$STREAMS" -t 4 "https://127.0.0.1:8443$URLPATH" > /tmp/h2l.$$ 2>&1
  local c1; c1=$(awk '{print $14+$15}' /proc/$pid/stat)
  kill $pid; wait $pid 2>/dev/null || true
  local n; n=$(awk '/^status codes:/{print $3}' /tmp/h2l.$$)
  local ins mis
  ins=$(awk -F, '/instructions/{print $1}' /tmp/perf.$$); mis=$(awk -F, '/L1-icache-load-misses/{print $1}' /tmp/perf.$$)
  awk -v c=$((c1 - c0)) -v hz="$HZ" -v n="$n" -v i="$ins" -v m="$mis" 'BEGIN{printf "%.2f %.0f %.1f", c/hz*1e6/n, i/n, m/n}'
}
declare -a bu bi bm nu ni nm
for r in $(seq 1 "$ROUNDS"); do
  read -r u i m <<<"$(one "$BASE")"; bu+=("$u"); bi+=("$i"); bm+=("$m")
  read -r u i m <<<"$(one "$NEW")"; nu+=("$u"); ni+=("$i"); nm+=("$m")
  echo "round $r: base ${bu[-1]}/${bi[-1]}/${bm[-1]}  new ${nu[-1]}/${ni[-1]}/${nm[-1]}" >&2
done
mean() { printf '%s\n' "$@" | awk '{s+=$1} END{printf "%.2f", s/NR}'; }
echo "h3 $URLPATH c=$CONNS m=$STREAMS, $ROUNDS rounds alternated"
echo "base  us/req ${bu[*]}  mean $(mean "${bu[@]}")  instructions/req $(mean "${bi[@]}")  L1i misses/req $(mean "${bm[@]}")"
echo "new   us/req ${nu[*]}  mean $(mean "${nu[@]}")  instructions/req $(mean "${ni[@]}")  L1i misses/req $(mean "${nm[@]}")"
```

- machine: x86_64, AMD Ryzen 9 9900X 12-Core Processor, Linux 6.12.113+deb13-amd64 (agensio-devbox:latest, uid 1000)
- commit: 634658fa + the working tree of access_log_exclude_paths (0.1.0-alpha.128)
- command: bench/tmp/excl-bench.sh (below), from the repository root in the devbox
- what: one binary, the access log on (/dev/null), one worker, wrk -t4 -c64 -d5s on / (1 KB), the config
  without the key and with access_log_exclude_paths = ["/api/notices/", "/healthz"] alternated five times;
  CPU us/req = the server's utime + stime over wrk's requests. A request not excluded pays the target's
  cleaning (normalize_target's plain-path copy) and two prefix compares.

```sh
#!/usr/bin/env bash
# What access_log_exclude_paths costs a site that has it, on a request it does not exclude: the same
# binary with the access log on (/dev/null), one config without the key and one with two paths,
# alternated, one worker, wrk -t4 -c64 -d5s on / (1 KB). Run in the devbox from the repo root.
set -euo pipefail
ROOT=$PWD
BIN=$ROOT/build/agensio
OUT=$ROOT/bench/tmp/excl
mkdir -p $OUT
sed "s#@WORKERS@#1#g; s#@BENCH@#$ROOT/bench#g; s#@SENDFILE_MIN@#48KB#g; s#@ACCESS_LOG@#/dev/null#g; s#@H3@##g" bench/agensio.toml > $OUT/plain.toml
awk '{print} /^listen = \["127.0.0.1:8080"\]$/ {print "access_log_exclude_paths = [\"/api/notices/\", \"/healthz\"]"}' $OUT/plain.toml > $OUT/excl.toml
grep -q access_log_exclude_paths $OUT/excl.toml
"$BIN" -t -c $OUT/excl.toml > /dev/null
TCK=$(getconf CLK_TCK)
ticks() { awk '{print $14 + $15}' /proc/$1/stat; }
run() {  # CONFIG -> us/req
  "$BIN" -c "$1" > $OUT/server.out 2>&1 & local pid=$!
  for _ in $(seq 1 50); do nc -z 127.0.0.1 8080 2>/dev/null && break; sleep 0.1; done
  wrk -t4 -c64 -d1s http://127.0.0.1:8080/ > /dev/null
  local a; a=$(ticks $pid)
  local w; w=$(wrk -t4 -c64 -d5s http://127.0.0.1:8080/)
  local b; b=$(ticks $pid)
  kill $pid; wait $pid 2>/dev/null || true
  local n; n=$(echo "$w" | awk '/requests in/ {print $1}')
  awk -v t=$((b - a)) -v hz=$TCK -v n=$n 'BEGIN {printf "%.3f", t / hz * 1e6 / n}'
}
echo "# access_log_exclude_paths cost, $(date -u +%FT%TZ), $(git rev-parse --short HEAD)+dirty, $(uname -sr)"
echo "| round | without the key us/req | with two paths us/req |"
echo "|---|---|---|"
for r in 1 2 3 4 5; do
  p=$(run $OUT/plain.toml); sleep 1
  e=$(run $OUT/excl.toml); sleep 1
  echo "| $r | $p | $e |"
done
```

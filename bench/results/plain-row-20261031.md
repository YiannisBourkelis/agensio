The script:

```bash
#!/usr/bin/env bash
# The http / row by hand: the A/B's own config (bench/tmp/ab-agensio.toml), base and new alternated.
set -uo pipefail
BASE=$(realpath "$1"); NEW=$(realpath "$2"); ROUNDS=${3:-6}; export LC_NUMERIC=C
cpu() { awk -v tck="$(getconf CLK_TCK)" '{ printf "%.3f", ($14+$15)/tck }' "/proc/$1/stat"; }
side() {
  (cd bench && exec "$2" -c tmp/ab-agensio.toml) > /dev/null 2>&1 & local PID=$!
  for _ in $(seq 1 50); do nc -z 127.0.0.1 8080 2>/dev/null && break; sleep 0.1; done
  wrk -t4 -c64 -d2s http://127.0.0.1:8080/ > /dev/null 2>&1
  local c0 c1 out reqs; c0=$(cpu $PID)
  out=$(wrk -t4 -c64 -d5s http://127.0.0.1:8080/); c1=$(cpu $PID)
  reqs=$(echo "$out" | awk '/requests in/{print $1}')
  echo "$1 $(awk -v a=$c0 -v b=$c1 -v n=$reqs 'BEGIN{printf "%.2f", (b-a)*1e6/n}')"
  kill $PID; wait $PID 2>/dev/null; sleep 0.3
}
for r in $(seq 1 $ROUNDS); do side base "$BASE"; side new "$NEW"; done
```

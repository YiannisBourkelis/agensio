`rows-by-hand.sh`:

```bash
#!/usr/bin/env bash
# A/B rows by hand: base and new alternated on one configuration, CPU us per request.
# usage: rows-by-hand.sh BASE NEW CONFIG ROUNDS URL...
set -uo pipefail
export LC_NUMERIC=C
BASE=$(realpath "$1"); NEW=$(realpath "$2"); CONF=$(realpath "$3"); ROUNDS=$4; shift 4
cpu() { awk -v tck="$(getconf CLK_TCK)" '{ printf "%.3f", ($14+$15)/tck }' "/proc/$1/stat"; }
"$PWD/build/agensio_upstream" -p 9100 -w 1 > /dev/null 2>&1 & UP=$!
sleep 0.3
side() {  # name binary url
  (cd bench && exec "$2" -c "$CONF") > /dev/null 2>&1 & local PID=$!
  for _ in $(seq 1 50); do nc -z 127.0.0.1 8080 2>/dev/null && break; sleep 0.1; done
  sleep 0.3
  wrk -t4 -c64 -d2s "$3" > /dev/null 2>&1
  local c0 c1 out reqs; c0=$(cpu $PID)
  out=$(wrk -t4 -c64 -d5s "$3"); c1=$(cpu $PID)
  reqs=$(echo "$out" | awk '/requests in/{print $1}')
  echo "$3 $1 $(awk -v a=$c0 -v b=$c1 -v n=$reqs 'BEGIN{printf "%.2f", (b-a)*1e6/n}')"
  kill $PID; wait $PID 2>/dev/null; sleep 0.3
}
for url in "$@"; do for r in $(seq 1 $ROUNDS); do side base "$BASE" "$url"; side new "$NEW" "$url"; done; done
kill $UP; wait $UP 2>/dev/null
```

`perf-row.sh`:

```bash
#!/usr/bin/env bash
# Instructions and cycles per request on one row, base and new alternated (perf stat on the server).
set -uo pipefail
export LC_NUMERIC=C
BASE=$(realpath "$1"); NEW=$(realpath "$2"); CONF=$(realpath "$3"); ROUNDS=$4; URL=$5
side() {
  (cd bench && exec "$2" -c "$CONF") > /dev/null 2>&1 & local PID=$!
  for _ in $(seq 1 50); do nc -z 127.0.0.1 8080 2>/dev/null && break; sleep 0.1; done
  sleep 0.3
  wrk -t4 -c64 -d2s "$URL" > /dev/null 2>&1
  perf stat -x, -e instructions,cycles,branch-misses,L1-icache-load-misses -p $PID -o /tmp/perf.$$ -- sleep 5 > /dev/null 2>&1 &
  local PP=$!
  local out; out=$(wrk -t4 -c64 -d5s "$URL"); wait $PP
  local reqs; reqs=$(echo "$out" | awk '/requests in/{print $1}')
  awk -F, -v n=$reqs -v who=$1 '{v[$3]=$1} END{printf "%s ins/req %.0f cyc/req %.0f ipc %.2f brmiss/req %.2f icmiss/req %.2f\n", who, v["instructions"]/n, v["cycles"]/n, v["instructions"]/v["cycles"], v["branch-misses"]/n, v["L1-icache-load-misses"]/n}' /tmp/perf.$$
  kill $PID; wait $PID 2>/dev/null; sleep 0.3
}
for r in $(seq 1 $ROUNDS); do side base "$BASE"; side new "$NEW"; done
```
